### 1. Instalar Python

sudo apt install python3-venv python3-full -y

### 2. Cria o ambiente virtual

python3 -m venv venv

### 3. Ativa o ambiente

source venv/bin/activate

### 4. Instalar dependências

pip install astropy astroquery
